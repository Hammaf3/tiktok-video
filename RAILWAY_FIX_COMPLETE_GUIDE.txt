ERROR A3: "ffmpeg: not found" OR "ffprobe: not found"
─────────────────────────────────────────────────────────────────────────

CAUSE: FFmpeg not installed on Railway despite nixpacks.toml

FIX: Create proper Dockerfile
cd /c/Users/Faraz/Desktop/tiktok\ video\ uploader

cat > Dockerfile << 'DOCKERFILE_END'
FROM python:3.12-slim

# Install FFmpeg
RUN apt-get update && \
    apt-get install -y ffmpeg && \
    rm -rf /var/lib/apt/lists/*

# Set working directory
WORKDIR /app

# Copy and install Python dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy application code
COPY . .

# Create temp directories
RUN mkdir -p tmp/yt2tik/output tmp/yt2tik/downloads

# Expose port (Railway sets $PORT)
EXPOSE 8080

# Run with gunicorn
CMD gunicorn integrated_app:app --bind 0.0.0.0:$PORT --workers 2 --timeout 300
DOCKERFILE_END

git add Dockerfile
git commit -m "Add Dockerfile for FFmpeg support"
git push origin master
